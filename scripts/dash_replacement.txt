const getRecentOrders = async (req, res) => {
  try {
    const { page = 1, limit = 5, status, startDate, endDate } = req.query;
    const pageNum = parseInt(page);
    const limitNum = parseInt(limit);
    const offset = (pageNum - 1) * limitNum;
    let whereClauses = [];
    let queryParams = [];

    if (status && status !== 'All') { whereClauses.push('o.status = ?'); queryParams.push(status.toLowerCase()); }
    if (startDate) { whereClauses.push('o.created_at >= ?'); queryParams.push(startDate); }
    if (endDate) { whereClauses.push('o.created_at <= ?'); queryParams.push(endDate + ' 23:59:59'); }

    const whereString = whereClauses.length > 0 ? 'WHERE ' + whereClauses.join(' AND ') : '';

    const countQuery = 'SELECT COUNT(*) as total FROM orders o ' + whereString;
    const [[countResult]] = await db.query(countQuery, queryParams);
    const total = countResult.total || 0;
    const totalPages = Math.ceil(total / limitNum);

    const dataQuery = 'SELECT o.*, u.name as user_name, u.phone as user_phone, a.phone as address_phone FROM orders o LEFT JOIN users u ON o.user_id = u.id LEFT JOIN addresses a ON o.address_id = a.id ' + whereString + ' ORDER BY o.created_at DESC LIMIT ? OFFSET ?';
    const [orders] = await db.query(dataQuery, [...queryParams, limitNum, offset]);

    for (let order of orders) {
      const [items] = await db.query('SELECT oi.*, p.name as product_name, p.description, p.price as current_price, c.name as category_name FROM order_items oi JOIN products p ON oi.product_id = p.id LEFT JOIN categories c ON p.category_id = c.id WHERE oi.order_id = ?', [order.id]);
      
      order.customer = {
        name: order.user_name || 'Customer',
        phone: order.address_phone || order.user_phone || '-'
      };
      
      if (items.length > 0) {
        order.product = {
          name: items[0].product_name,
          shortDesc: items[0].description ? items[0].description.substring(0, 30) + '...' : '',
          category: items[0].category_name || 'General',
          price: items[0].price || items[0].current_price,
          qty: items[0].quantity,
          packageSize: '-'
        };
      } else {
        order.product = { name: 'No items', shortDesc: '', category: '-', price: 0, qty: 0, packageSize: '-' };
      }
    }

    res.status(200).json({
      success: true,
      data: {
        orders,
        pagination: { page: pageNum, limit: limitNum, total, totalPages }
      }
    });
  } catch (error) {
    console.error('Error fetching recent orders:', error);
    res.status(500).json({ success: false, message: 'Server error fetching recent orders.' });
  }
};
